cmake_minimum_required(VERSION 3.13)

# Pico SDK importieren
include(pico_sdk_import.cmake)

project(pico_uf2loader_project C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# SDK initialisieren
pico_sdk_init()

# Das ausführbare Programm definieren
add_executable(main
    main.c
    display.c
)

# Standard-Bibliotheken des Pico verknüpfen
target_link_libraries(main 
    pico_stdlib 
    hardware_spi
    hardware_gpio
)

# Aktiviert die Erstellung der .uf2 Datei
pico_add_extra_outputs(main)
